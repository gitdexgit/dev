# cal

FLAGS
  -m        Week starts on Monday
  -w        Show week numbers
  -n [N]    Show N months ahead
  -S        Center months around today
  (Tip: combine flags like -wm)

COMMANDS
  cal               This month
  cal -wm           This month (Mon start + week numbers)
  cal -wm 2026      Entire year 2026
  cal -wm -n 4      Next 4 months
  cal -wm -n 4 -S   4 months centered on today
  cal -wm3          3 months (last, this, next)
